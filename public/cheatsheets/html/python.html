🏗️ Delta Live Tables (DLT) Declarative Pipelines DLT 
import dlt
from pyspark.sql import functions as F

# Streaming Ingestion Table
@dlt.table(
    name="bronze_txns",
    comment="Raw transaction stream",
    table_properties={"quality": "bronze"}
)
def bronze_txns():
    return (spark.readStream.format("cloudFiles")
            .option("cloudFiles.format", "json")
            .load("abfss://landing@adls.dfs.core.windows.net/txns/"))

# Silver Table with Data Quality Expectations
@dlt.table(comment="Validated transactions")
@dlt.expect_or_drop("valid_id", "txn_id IS NOT NULL")
@dlt.expect_or_fail("valid_amt", "amount > 0.0")
def silver_txns():
    return dlt.read_stream("bronze_txns").filter(F.col("status") != "REJECT")
💡 DLT Expectations: expect (logs metric), expect_or_drop (drops bad rows), expect_or_fail (fails pipeline on violation).
🔄 CDC Ingestion with dlt.apply_changes() SCD Type 1 & 2 
import dlt

# Automatic SCD Type 2 History Tracking
dlt.create_streaming_table("gold_customers_scd2")

dlt.apply_changes(
    target="gold_customers_scd2",
    source="silver_customer_cdc",
    keys=["customer_id"],
    sequence_by="updated_timestamp",
    apply_as_deletes="operation = 'DELETE'",
    except_column_list=["operation"],
    stored_as_scd_type=2 # Auto creates __START_AT and __END_AT
)
  • stored_as_scd_type=1 overwrites records in place (latest snapshot).
  • stored_as_scd_type=2 manages effective date ranges and active flags out-of-the-box!

📊 Structured Logging & Observability Observability 
import json, logging
from datetime import datetime

class StructuredPipelineLogger:
    def __init__(self, pipeline_name: str):
        self.name = pipeline_name

    def log_event(self, event_name: str, records: int, duration_sec: float):
        payload = {
            "timestamp": datetime.utcnow().isoformat(),
            "pipeline": self.name,
            "event": event_name,
            "records_processed": records,
            "duration_sec": duration_sec
        }
        print(json.dumps(payload)) # Ingested into Azure Log Analytics

audit = StructuredPipelineLogger("SEPA_Payment_Processor")
audit.log_event("STAGE_COMPLETED", records=845000, duration_sec=14.2)
🔄 Concurrent Extraction & Exponential Retries Resilience 
from concurrent.futures import ThreadPoolExecutor
from tenacity import retry, stop_after_attempt, wait_exponential

@retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=2, max=10))
def fetch_api_page(page_id: int) -> dict:
    # Resilient REST API caller
    return {"page": page_id, "data": [1, 2, 3]}

# Parallelize lightweight API calls on Driver before Spark processing
with ThreadPoolExecutor(max_workers=8) as executor:
    results = list(executor.map(fetch_api_page, range(1, 50)))
⚡ Thread Concurrency: Use Python ThreadPoolExecutor on the driver for I/O-bound REST calls, then parallelize the dataset with spark.createDataFrame().
🛡️ Type Annotations & Schema Contracts Data Quality 
from pyspark.sql import DataFrame, Column
from typing import Callable

def apply_curation(
    df: DataFrame,
    primary_key: str,
    business_rule: Callable[[Column], Column]
) -> DataFrame:
    assert primary_key in df.columns, f"Missing required key: {primary_key}"
    return df.filter(business_rule(df[primary_key]))
  • Enforce PEP 484 type hints across modular libraries.
  • Pre-flight assertions catch schema mismatch errors in milliseconds before distributed stages start.
🧪 Unit Testing PySpark Pipelines in CI/CD Testing 
import pytest
from pyspark.testing.utils import assertDataFrameEqual

def test_filter_active(spark):
    input_data = [(1, "ACTIVE"), (2, "CANCELLED")]
    input_df = spark.createDataFrame(input_data, ["id", "status"])
    
    expected_data = [(1, "ACTIVE")]
    expected_df = spark.createDataFrame(expected_data, ["id", "status"])
    
    result_df = input_df.filter("status = 'ACTIVE'")
    # Spark 3.5+ built-in test assertion
    assertDataFrameEqual(result_df, expected_df)
💡 pyspark.testing: Use assertDataFrameEqual and assertSchemaEqual in pytest to validate transformations without third-party libraries.
